```javascript
// 改后：
function deleteLogEntry(id) {
    // 找到被删的主线条目
    const target = logs.find(l => l.id === id);
    // 删除主线条目标
    logs = logs.filter(l => l.id !== id);
    // 如果被删的是主线条目（非并行），也删除挂在其下的并行子条目
    if (target && !target.parallel) {
        logs = logs.filter(l => !(l.parallel && l.parentId === target.id));
    }
    localStorage.setItem('v9_logs', JSON.stringify(logs));
    renderAll();
}
```

**新增 `executeSplit()` 函数：**

```javascript
function executeSplit(parentLog, l1, l2) {
    const pStart = parentLog.startTime;
    const pEnd = parentLog.endTime || (pStart + (parentLog.duration || 60) * 60000);
    const splitStart = clampTime(parseTimeFromInput('ps', pStart), pStart, pEnd);
    const splitEnd = clampTime(parseTimeFromInput('pe', pStart), pStart, pEnd);
    
    if (splitEnd - splitStart < 60000) { // 至少1分钟
        closeDrawer();
        return;
    }
    
    // 从 logs 里移除原来的条目
    const idx = logs.indexOf(parentLog);
    if (idx === -1) { closeDrawer(); return; }
    logs.splice(idx, 1);
    
    const cat = getCat(l1);
    const color = cat?.color || '#cbd5e1';
    const newEntries = [];
    
    // 前段（如果 splitStart > pStart）
    if (splitStart > pStart) {
        newEntries.push({
            ...parentLog,
            id: Date.now() + Math.random(),
            startTime: pStart,
            endTime: splitStart,
            duration: Math.round((splitStart - pStart) / 60000),
        });
    }
    
    // 中段（切割出来的新活动）
    newEntries.push({
        id: Date.now() + Math.random() + 1,
        startTime: splitStart,
        endTime: splitEnd,
        duration: Math.round((splitEnd - splitStart) / 60000),
        l1, l2: l2 || '',
        note: document.getElementById('drawer-note').value || '',
        color,
        tag: '',
    });
    
    // 后段（如果 splitEnd < pEnd）
    if (splitEnd < pEnd) {
        newEntries.push({
            ...parentLog,
            id: Date.now() + Math.random() + 2,
            startTime: splitEnd,
            endTime: pEnd,
            duration: Math.round((pEnd - splitEnd) / 60000),
        });
    }
    
    // 插入新条目（保持顺序）
    logs.splice(idx, 0, ...newEntries);
    
    // 自动合并相邻同活动
    mergeAdjacentSameActivity();
    
    localStorage.setItem('v9_logs', JSON.stringify(logs));
    closeDrawer();
    renderAll();
}

function mergeAdjacentSameActivity() {
    const merged = [];
    for (const log of logs) {
        const last = merged[merged.length - 1];
        if (last && last.l1 === log.l1 && last.l2 === log.l2 && last.endTime === log.startTime) {
            // 合并：延长 last
            last.endTime = log.endTime;
            last.duration = Math.round((last.endTime - last.startTime) / 60000);
        } else {
            merged.push({ ...log });
        }
    }
    logs = merged;
}
```
